5. DATABASE SCHEMA SUMMARY
================================================================================

DATABASE 1: applicantdb.sqlite (Applicant Profiles)

Table: applicants
- id (TEXT PRIMARY KEY)
- first_name, last_name, email, phone, ssn
- date_of_birth, address, city, state, zip_code
- employment_status, employer_id, annual_income
- enrichment_data (JSON: credit, employment, identity enrichment)
- validation_status (valid/invalid/pending_validation)
- created_at, updated_at
Indexes: email, ssn

Table: credit_history
- id (INTEGER PRIMARY KEY AUTOINCREMENT)
- applicant_id (FOREIGN KEY)
- credit_score, total_debt, available_credit
- payment_history (JSON)
- enrichment_source, updated_by, last_updated
Indexes: applicant_id, last_updated

DATABASE 2: employers.sqlite (Employment Data)

Table: employers
- id (TEXT PRIMARY KEY)
- name, industry, headquarters_state, employee_count
- verified, created_at
Indexes: name

Table: employee_records
- id (INTEGER PRIMARY KEY AUTOINCREMENT)
- employer_id, applicant_id (FOREIGN KEYS)
- employee_id, job_title, salary, start_date
- current_status, verified_at
Indexes: employer_id, applicant_id

DATABASE 3: audit.sqlite (Audit Trail & Webhooks)

Table: audit_log
- id (TEXT PRIMARY KEY)
- event_type, applicant_id, entity_type, entity_id, action
- old_values (JSON), new_values (JSON)
- user_id, ip_address, timestamp, status
Indexes: applicant_id, timestamp

Table: webhooks
- id (TEXT PRIMARY KEY)
- url, event_type, is_active
- created_at, last_triggered
Indexes: event_type

Table: webhook_logs
- id (TEXT PRIMARY KEY)
- webhook_id (FOREIGN KEY)
- event_type, payload (JSON)
- status_code, response_text, triggered_at
Indexes: webhook_id
