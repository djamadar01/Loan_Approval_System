🖥️  WHICH TERMINAL TO OPEN?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

ANY TERMINAL WILL WORK:
✓ Terminal (Ubuntu default)
✓ Konsole
✓ xterm
✓ gnome-terminal

KEYBOARD SHORTCUT (Fastest):
  Ctrl + Alt + T

BUTTON CLICK:
  Click Terminal icon in taskbar (bottom left usually)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
📋 INSTALLATION STEPS (Copy-Paste)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

STEP 1: Open Terminal
  → Press Ctrl + Alt + T

STEP 2: Install Streamlit (Copy-Paste This)
  
  pip install streamlit
  
  Then press ENTER

STEP 3: Wait for Installation
  You'll see:
  
    Collecting streamlit...
    Installing...
    Successfully installed streamlit
  
  ✅ Installation done!

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
🚀 RUN THE UI (Copy-Paste These Commands)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

COMMAND 1: Go to Project Directory
  
  cd /home/ubuntu/Desktop/demo
  
  Press ENTER

COMMAND 2: Start the UI
  
  streamlit run app.py
  
  Press ENTER

WHAT HAPPENS:
  • Terminal shows messages
  • Browser opens automatically
  • You see: http://localhost:8501
  • Form appears in browser

✅ UI IS RUNNING!

❌ IF SOMETHING GOES WRONG
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

"streamlit: command not found"
→ Try: python3 -m streamlit run app.py

"Port 8501 already in use"
→ Try: streamlit run app.py --server.port 8502

"ModuleNotFoundError"
→ Try: pip install pandas numpy plotly

"Permission denied"
→ Try: pip install --user streamlit
